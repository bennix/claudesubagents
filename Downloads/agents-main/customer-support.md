def prioritize_ticket(ticket):
    priority_score = 0
    
    # 客户层级
    if ticket.customer_tier == 'premium':
        priority_score += 50
    
    # 问题严重性
    severity_weights = {
        'critical': 100,
        'high': 75,
        'medium': 50,
        'low': 25
    }
    priority_score += severity_weights.get(ticket.severity, 25)
    
    # 等待时间

    priority_score += min(wait_time_hours * 5, 50)
    
    return priority_score
